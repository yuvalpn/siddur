import os import re import json import zipfile import requests from google.colab import files BASE_URL = "https://www.sefaria.org/api/texts/" HEADERS = {"User-Agent": "SmartSiddurApp/1.0"} def clean_hebrew_text(raw_text): """ניקוי תגיות HTML, הערות ורווחים עודפים""" if isinstance(raw_text, list): return [clean_hebrew_text(t) for t in raw_text if t] if not isinstance(raw_text, str): return "" text = re.sub(r'<[^>]+>', '', raw_text) text = re.sub(r'[\(\[].*?[\)\]]', '', text) text = re.sub(r'\s+', ' ', text).strip() return text def fetch_sefaria(ref): """שליפת טקסט מספריא לפי נתיב מדויק""" print(f"מושך: {ref}...") url = f"{BASE_URL}{requests.utils.quote(ref)}?context=0&commentary=0" try: res = requests.get(url, headers=HEADERS, timeout=15) if res.status_code == 200: data = res.json() return clean_hebrew_text(data.get("he", "")) else: print(f"שגיאה {res.status_code} עבור {ref}") return "" except Exception as e: print(f"שגיאת תקשורת ב-{ref}: {e}") return "" os.makedirs("data", exist_ok=True) #
